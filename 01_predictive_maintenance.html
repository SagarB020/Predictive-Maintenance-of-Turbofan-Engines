    group = group.sort_values("cycle").reset_index(drop=True)

        y_seq.append(rul_values[i+window_size-1])

X_seq = np.asarray(X_seq, dtype=np.float32)

train a simple LSTM in PyTorch¶

In [20]:
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

input_dim = X_seq.shape[2]

class LSTMRegressor(nn.Module):
    def __init__(self, input_dim):
        super().__init__()
        self.lstm = nn.LSTM(
            input_size=input_dim,
            hidden_size=64,
            num_layers=2,
            dropout=0.2,
            batch_first=True
        )
        self.fc1 = nn.Linear(64, 32)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(32, 1)

    def forward(self, x):
        out, _ = self.lstm(x)
        out = out[:, -1, :]
        out = self.fc1(out)
        out = self.relu(out)
        out = self.fc2(out)
        return out.squeeze(-1)

# Data split
idx = np.arange(len(X_seq))
np.random.seed(42)
np.random.shuffle(idx)

split = int(0.8 * len(X_seq))
train_idx = idx[:split]
val_idx = idx[split:]

X_train = torch.tensor(X_seq[train_idx], dtype=torch.float32)
y_train = torch.tensor(y_seq[train_idx], dtype=torch.float32)
X_val = torch.tensor(X_seq[val_idx], dtype=torch.float32)
y_val = torch.tensor(y_seq[val_idx], dtype=torch.float32)

train_loader = DataLoader(TensorDataset(X_train, y_train), batch_size=64, shuffle=True)
val_loader = DataLoader(TensorDataset(X_val, y_val), batch_size=64, shuffle=False)

model_lstm = LSTMRegressor(input_dim)
criterion = nn.MSELoss()
optimizer = optim.Adam(model_lstm.parameters(), lr=1e-3)

best_val_loss = float("inf")
patience = 5
counter = 0

for epoch in range(15):
    model_lstm.train()
    train_loss = 0.0

    for xb, yb in train_loader:
        optimizer.zero_grad()
        pred = model_lstm(xb)
        loss = criterion(pred, yb)
        loss.backward()
        optimizer.step()
        train_loss += loss.item() * xb.size(0)

    train_loss /= len(train_loader.dataset)

    model_lstm.eval()
    val_loss = 0.0
    with torch.no_grad():
        for xb, yb in val_loader:
            pred = model_lstm(xb)
            loss = criterion(pred, yb)
            val_loss += loss.item() * xb.size(0)

    val_loss /= len(val_loader.dataset)

    print(f"Epoch {epoch+1}/15 | train_loss={train_loss:.4f} | val_loss={val_loss:.4f}")

    if val_loss < best_val_loss:
        best_val_loss = val_loss
        counter = 0
    else:
        counter += 1

    if counter >= patience:
        print("Early stopping due to no improvement.")
        break

lstm_model_path = os.path.join(models_dir, "lstm_model.pt")
torch.save(model_lstm.state_dict(), lstm_model_path)
print("Saved LSTM model:", lstm_model_path)
